\subsection*{1. Short Answer Questions (30 points)}
\question
\begin{parts}
	\part[4] Profit-maximizing price:
	\begin{solution}
	$\frac{P - MC}{P} = \frac{1}{|\varepsilon|} = \frac{1}{2} \implies P - 30 = 0.5P \implies P^* = \$60$
	\end{solution}

	\part[4] Interpretation of $\alpha$:
	\begin{solution}
	$\alpha$ is the price coefficient in \underline{dollars}: if price goes up by \$1, utility decreases by $\alpha$ dollars.
	\end{solution}

	\part[4] Adding 5 to all goods (including outside option):
	\begin{solution}
	True. $s_j = \frac{e^{\delta_j + 5}}{e^5 + \sum_k e^{\delta_k + 5}} = \frac{e^5 \cdot e^{\delta_j}}{e^5(1 + \sum_k e^{\delta_k})} = \frac{e^{\delta_j}}{1 + \sum_k e^{\delta_k}}$
	\end{solution}

	\part[4] BLP instruments relevance:
	\begin{solution}
	More/closer competitors $\rightarrow$ more competition $\rightarrow$ lower prices.
	\end{solution}

	\part[4] Higher-priced products typically more elastic:
	\begin{solution}
	True. $\eta_{jj} \approx \alpha p_j$ when $s_j$ small. Since $\alpha < 0$, higher $p_j$ $\rightarrow$ more elastic.
	\end{solution}

	\part[3] DWL under average cost pricing:
	\begin{solution}
	False. $P = AC > MC$ $\Rightarrow$ consumers with WTP $\in (MC, AC)$ don't buy $\Rightarrow$ DWL $> 0$.
	\end{solution}

	\part[4] False CS formula (two goods example):
	\begin{solution}
	Incorrect. Correct formula is $CS = \frac{1}{|\alpha|} \ln(1 + e^{\delta_1} + e^{\delta_2}) = \frac{\ln(1 + 2e^2)}{|\alpha|} \approx \frac{2.76}{|\alpha|}$.

	The proposed formula gives $\frac{2e^2}{|\alpha|} \approx \frac{14.8}{|\alpha|}$---missing the log.
	\end{solution}

	\part[3] DWL under perfect price discrimination:
	\begin{solution}
	True. Sells to all with WTP $\geq$ MC $\Rightarrow$ efficient quantity $\Rightarrow$ DWL = 0.
	\end{solution}
\end{parts}

\subsection*{2. Demand Estimation (30 points)}
\question
\begin{parts}
	\part[5] Define $\delta_{jt}$ and derive Berry inversion:
	\begin{solution}
	$\delta_{jt} = \beta \cdot \text{mpg}_j + \alpha \cdot p_{jt} + \xi_{jt}$ (part of utility common to all consumers).

	Berry inversion: $\delta_{jt} = \ln(s_{jt}) - \ln(s_{0t})$
	\end{solution}

	\part[5] Regression equation:
	\begin{solution}
	$\ln(s_{jt}) - \ln(s_{0t}) = \beta \cdot \text{mpg}_j + \alpha \cdot p_{jt} + \xi_{jt}$

	Dependent variable: $\ln(s_{jt}) - \ln(s_{0t})$. Regressors: $\text{mpg}_j$, $p_{jt}$. Error: $\xi_{jt}$.
	\end{solution}

	\part[5] OLS bias:
	\begin{solution}
	$\text{Cov}(p_{jt}, \xi_{jt}) > 0$: high-quality cars command higher prices.

	Upward bias (less negative $\hat{\alpha}$): OLS attributes high demand for expensive cars to low price sensitivity rather than unobserved quality.
	\end{solution}

	\part[5] Hausman instruments:
	\begin{solution}
	If $\xi_{jt}$ is correlated across markets (e.g., brand reputation), then other-market prices are correlated with own-market $\xi_{jt}$, violating the exclusion restriction.
	\end{solution}

	\part[5] Advantage of demographic interactions:
	\begin{solution}
	Allows heterogeneous price sensitivity across consumers, partially addressing IIA: substitution patterns depend on which consumers buy each product.
	\end{solution}

	\part[5] Corolla-Odd vs. Corolla-Even:
	\begin{solution}
	Model predicts CS increases: $CS = \frac{1}{|\alpha|} \ln(1 + \sum_j \exp(\delta_j))$ rises when products are added.

	Should we believe it? No. The ``products'' are physically identical---consumers don't care about license plates. This is an example of the ``red-bus blue-bus'' problem.
	\end{solution}
\end{parts}

\subsection*{3. Two-Part Tariffs (20 points)}
\question
\begin{parts}
	\part[5] Profit as a function of $F$ and $p$:
	\begin{solution}
	$q = 20 - 2p$, choke price $p = 10$.

	$CS(p) = \frac{1}{2}(20-2p)(10-p) = (10-p)^2$, so max fee $F = (10-p)^2$.

	$\Pi = 100 \left[ (10-p)^2 + (p-2)(20-2p) \right]$
	\end{solution}

	\part[5] Show $p^* = c = 2$:
	\begin{solution}
	$\frac{d\pi}{dp} = -2(10-p) + (20-2p) - 2(p-2) = 4 - 2p$

	$4 - 2p = 0 \Rightarrow p^* = 2 = c$
	\end{solution}

	\part[10] Comparing two pricing strategies:
	\begin{solution}
	At $p = 0$: $CS_H = \frac{1}{2}(10)(10) = 50$, $CS_L = \frac{1}{2}(5)(5) = 12.5$.

	(a) $F = 50$: Heavy join ($CS_H = 50 \geq F$), light don't ($CS_L = 12.5 < F$). Profit $= 50 \times 50 = \$2,500$.

	(b) $F = 12.5$: Both join. Profit $= 150 \times 12.5 = \$1,875$.

	Strategy (a) is better.
	\end{solution}
\end{parts}

\subsection*{4. Self-Selection (20 points)}
\question
\begin{parts}
	\part[4] Perfect price discrimination:
	\begin{solution}
	Businesses: $P_P = \$400$. Students: $P_B = \$100$.

	Profit $= 40(370) + 60(70) = \$19,000$
	\end{solution}

	\part[4] IC constraint for businesses:
	\begin{solution}
	$400 - P_P \geq 150 - P_B \implies P_P - P_B \leq 250$
	\end{solution}

	\part[6] Why $P_P = \$380$, $P_B = \$100$ is not optimal:
	\begin{solution}
	Business surplus from Pro: $400 - 380 = 20$. From Basic: $150 - 100 = 50$.

	Since $20 < 50$, businesses prefer Basic. IC violated: $P_P - P_B = 280 > 250$.
	\end{solution}

	\part[6] Self-selection prices:
	\begin{solution}
	Binding: Student IR ($P_B = 100$) and Business IC ($P_P = P_B + 250 = 350$).

	Check student IC: $100 - 100 = 0 \geq 120 - 350 = -230$
	Profit $= 40(320) + 60(70) = \$17,000$
	\end{solution}
\end{parts}
