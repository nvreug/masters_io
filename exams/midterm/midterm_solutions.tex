\subsection*{1. Short Answer Questions (30 points)}
\question
\begin{parts}
	\part[3] Lerner index with $\varepsilon = -2$:
	\begin{solution}
	$L = \frac{P - MC}{P} = \frac{1}{|\varepsilon|} = \frac{1}{2} = \boxed{0.5}$
	\end{solution}

	\part[3] Berry inversion formula:
	\begin{solution}
	$\boxed{\ln(s_{jt}) - \ln(s_{0t}) = \delta_{jt} + \alpha p_{jt}}$

	Or equivalently: $\delta_{jt} = \ln(s_{jt}) - \ln(s_{0t}) - \alpha p_{jt}$
	\end{solution}

	\part[3] Demographics and IIA:
	\begin{solution}
	\textbf{False.} Demographics help at the aggregate level by creating different consumer segments, but for any individual consumer type, IIA still holds. Only random coefficients (mixed logit) fully address IIA by allowing idiosyncratic preferences.
	\end{solution}

	\part[3] Interpretation of $\alpha$:
	\begin{solution}
	$\alpha$ is the marginal disutility of price (negative because higher prices reduce utility). It can be interpreted as the negative of the marginal utility of income, relating utils to dollars. It determines price sensitivity and is used to convert utility changes to dollar-equivalent consumer surplus.
	\end{solution}

	\part[3] Cost pass-through with linear demand:
	\begin{solution}
	\textbf{False.} With linear demand $P = a - bQ$, the monopolist passes through only 50\% of cost increases. If $MC$ increases by \$1, the optimal price increases by \$0.50.

	(Derivation: FOC gives $P = \frac{a + MC}{2}$, so $\frac{dP}{dMC} = \frac{1}{2}$)
	\end{solution}

	\part[3] Two-part tariff optimal price:
	\begin{solution}
	\textbf{True.} With homogeneous consumers, the optimal two-part tariff sets $p = MC$ to maximize total surplus (eliminate deadweight loss), then extracts all consumer surplus via the fixed fee $F = CS$.
	\end{solution}

	\part[3] Why BLP instruments work:
	\begin{solution}
	BLP instruments (characteristics of other products) affect the firm's competitive environment, which shifts the firm's pricing behavior. Products facing more similar competitors charge lower prices. These characteristics are correlated with price (relevance) but uncorrelated with unobserved demand shocks (exclusion)---other products' characteristics don't directly affect own-product demand.
	\end{solution}

	\part[3] Which IC constraint binds:
	\begin{solution}
	\textbf{False.} The binding IC constraint is that the \textbf{high type must not prefer the low-type product}. The high type is tempted to ``mimic'' the low type to get a lower price. The low type's IC typically doesn't bind because the high-quality product is priced too high for them anyway.
	\end{solution}

	\part[3] Versioning:
	\begin{solution}
	\textbf{Versioning} is creating multiple versions of a product at different quality levels to induce consumer self-selection. Often involves ``damaging'' a product (adding restrictions, removing features) to make it less attractive to high-value consumers. Examples: software lite vs. pro, airline economy vs. business, student editions.
	\end{solution}

	\part[3] Consumer surplus formula:
	\begin{solution}
	\textbf{True.} $CS = \frac{1}{|\alpha|} \ln(1 + \sum_j \exp(v_j))$ where $v_j = \delta_{jt} + \alpha p_{jt}$.

	The inclusive value is $IV = \ln(1 + \sum_j \exp(v_j))$, and $CS = IV / |\alpha|$.
	\end{solution}
\end{parts}

\subsection*{2. Demand Estimation (30 points)}
\question
\begin{parts}
	\part[8] Compute $v_j$ and market shares:
	\begin{solution}
	$v_j = \delta_{jt} + \alpha p_{jt}$ with $\alpha = -0.5$:

	$v_1 = 4.0 + (-0.5)(20) = 4.0 - 10.0 = -6.0$

	$v_2 = 3.0 + (-0.5)(15) = 3.0 - 7.5 = -4.5$

	$v_3 = 2.0 + (-0.5)(10) = 2.0 - 5.0 = -3.0$

	$\exp(v_j)$: $e^{-6.0} = 0.00248$, $e^{-4.5} = 0.0111$, $e^{-3.0} = 0.0498$

	Denominator: $1 + 0.00248 + 0.0111 + 0.0498 = 1.0634$

	Market shares:

	$s_1 = 0.00248 / 1.0634 = \boxed{0.0023}$ (0.23\%)

	$s_2 = 0.0111 / 1.0634 = \boxed{0.0104}$ (1.04\%)

	$s_3 = 0.0498 / 1.0634 = \boxed{0.0468}$ (4.68\%)

	$s_{0} = 1 / 1.0634 = 0.940$ (94.0\%)
	\end{solution}

	\part[6] Own-price elasticities:
	\begin{solution}
	$\eta_{jj} = \alpha p_{j} (1 - s_{j})$

	$\eta_{11} = (-0.5)(20)(1 - 0.0023) = (-0.5)(20)(0.9977) = \boxed{-9.98}$

	$\eta_{22} = (-0.5)(15)(1 - 0.0104) = (-0.5)(15)(0.9896) = \boxed{-7.42}$

	$\eta_{33} = (-0.5)(10)(1 - 0.0468) = (-0.5)(10)(0.9532) = \boxed{-4.77}$

	\textbf{Product 3 has the most inelastic demand} ($|\eta_{33}| = 4.77$), driven by its lower price.
	\end{solution}

	\part[6] OLS bias:
	\begin{solution}
	OLS is biased because price $p_j$ is correlated with the error term (unobserved quality $\xi_j$).

	\textbf{Mechanism:} Firms with higher unobserved quality (better brand, taste, advertising) can charge higher prices. This creates positive correlation: $\text{Cov}(p_j, \xi_j) > 0$.

	\textbf{Direction:} Upward bias (less negative $\alpha$). OLS interprets high prices with high shares as ``consumers aren't price sensitive'' when really high $\xi_j$ is driving both.

	\textbf{Solution:} Use instrumental variables (cost shifters, BLP instruments) that are correlated with price but uncorrelated with $\xi_j$.
	\end{solution}

	\part[10] Consumer surplus calculation:
	\begin{solution}
	$IV = \ln(1 + \sum_j \exp(v_j)) = \ln(1.0634) = 0.0615$

	$CS = \frac{IV}{|\alpha|} = \frac{0.0615}{0.5} = \boxed{\$0.123}$ per consumer

	\textbf{Without product 3:}

	$IV' = \ln(1 + e^{-6.0} + e^{-4.5}) = \ln(1 + 0.00248 + 0.0111) = \ln(1.0136) = 0.0135$

	$CS' = \frac{0.0135}{0.5} = \$0.027$

	$\Delta CS = 0.027 - 0.123 = \boxed{-\$0.096}$ per consumer

	Consumers lose about 9.6 cents per person from removing product 3.
	\end{solution}
\end{parts}

\subsection*{3. Monopoly Pricing (20 points)}
\question
\begin{parts}
	\part[5] Profit-maximizing quantity and price:
	\begin{solution}
	Inverse demand: $P = 100 - 2Q$

	Revenue: $R = PQ = (100 - 2Q)Q = 100Q - 2Q^2$

	$MR = \frac{dR}{dQ} = 100 - 4Q$

	Set $MR = MC$: $100 - 4Q = 20$

	$4Q = 80 \Rightarrow \boxed{Q^* = 20}$

	$\boxed{P^* = 100 - 2(20) = \$60}$
	\end{solution}

	\part[5] Lerner index and elasticity verification:
	\begin{solution}
	Lerner index: $L = \frac{P - MC}{P} = \frac{60 - 20}{60} = \frac{40}{60} = \boxed{\frac{2}{3}}$

	To verify $L = 1/|\varepsilon|$, compute elasticity at optimal price:

	From $P = 100 - 2Q$, we get $Q = 50 - 0.5P$

	$\frac{dQ}{dP} = -0.5$

	$\varepsilon = \frac{dQ}{dP} \cdot \frac{P}{Q} = (-0.5) \cdot \frac{60}{20} = -1.5$

	$\frac{1}{|\varepsilon|} = \frac{1}{1.5} = \frac{2}{3} = L$ \quad \checkmark
	\end{solution}

	\part[5] Effect of cost increase:
	\begin{solution}
	With $MC = 30$:

	$MR = MC$: $100 - 4Q = 30$

	$4Q = 70 \Rightarrow Q^* = 17.5$

	$P^* = 100 - 2(17.5) = \$65$

	Price increase: $65 - 60 = \$5$

	Cost increase: $30 - 20 = \$10$

	\textbf{Pass-through rate:} $\frac{\$5}{\$10} = \boxed{50\%}$

	With linear demand, monopolists pass through only half of cost increases.
	\end{solution}

	\part[5] Cournot duopoly:
	\begin{solution}
	With 2 symmetric firms, $Q = q_1 + q_2$. Each firm's FOC:

	$\pi_i = (100 - 2Q)q_i - 20q_i = (100 - 2q_i - 2q_j - 20)q_i$

	FOC: $\frac{d\pi_i}{dq_i} = 100 - 4q_i - 2q_j - 20 = 0$

	Best response: $q_i = \frac{80 - 2q_j}{4} = 20 - 0.5q_j$

	Symmetry: $q_1 = q_2 = q^*$

	$q^* = 20 - 0.5q^* \Rightarrow 1.5q^* = 20 \Rightarrow q^* = \frac{40}{3} \approx 13.33$

	$Q = 2q^* = \frac{80}{3} \approx 26.67$

	$P = 100 - 2(\frac{80}{3}) = 100 - \frac{160}{3} = \frac{140}{3} \approx \boxed{\$46.67}$

	\textbf{Comparison:} Cournot price (\$46.67) $<$ Monopoly price (\$60). Competition lowers price.
	\end{solution}
\end{parts}

\subsection*{4. Self-Selection (20 points)}
\question
\begin{parts}
	\part[4] Perfect price discrimination:
	\begin{solution}
	Charge each type their full WTP for their preferred version:

	Businesses: $P_P = \$400$, sell Pro

	Students: $P_B = \$100$, sell Basic

	Profit: $40(400 - 50) + 60(100 - 20) = 40(350) + 60(80) = 14,000 + 4,800 = \boxed{\$18,800}$
	\end{solution}

	\part[4] IC constraint for businesses:
	\begin{solution}
	Businesses must prefer Pro to Basic:

	$WTP_P^{Bus} - P_P \geq WTP_B^{Bus} - P_B$

	$400 - P_P \geq 150 - P_B$

	$\boxed{P_P - P_B \leq 250}$

	(Or equivalently: $P_P \leq P_B + 250$)
	\end{solution}

	\part[6] Self-selection prices:
	\begin{solution}
	Constraints:
	\begin{itemize}
	\item IC for businesses: $400 - P_P \geq 150 - P_B$, i.e., $P_P \leq P_B + 250$
	\item IC for students: $100 - P_B \geq 120 - P_P$, i.e., $P_P \geq P_B + 20$
	\item IR for students: $P_B \leq 100$
	\item IR for businesses: $P_P \leq 400$
	\end{itemize}

	The binding constraints are:
	\begin{itemize}
	\item Student IR: $P_B = 100$ (extract all student surplus)
	\item Business IC: $P_P = P_B + 250 = 350$ (businesses just indifferent)
	\end{itemize}

	Check student IC: $100 - 100 = 0 \geq 120 - 350 = -230$ \checkmark

	Profit: $40(350 - 50) + 60(100 - 20) = 40(300) + 60(80) = 12,000 + 4,800 = \boxed{\$16,800}$

	(Lost \$2,000 compared to perfect discrimination due to business IC constraint)
	\end{solution}

	\part[6] Effect of ``damaging'' Basic:
	\begin{solution}
	Damaging Basic (removing features) \textbf{lowers businesses' WTP for Basic}.

	Suppose damage reduces business WTP for Basic from \$150 to \$50.

	New business IC: $400 - P_P \geq 50 - P_B$, i.e., $P_P \leq P_B + 350$

	Now the firm can charge $P_P = 100 + 350 = \$450$... but this exceeds business WTP of \$400.

	So charge $P_P = \$400$ (binding IR), $P_B = \$100$ (still extract all student surplus)

	Profit: $40(400 - 50) + 60(100 - 20) = 14,000 + 4,800 = \$18,800$

	\textbf{Economic logic:} By making Basic less attractive to high-value businesses, the firm relaxes the IC constraint and can charge businesses closer to their full WTP (\$400 vs. \$350 before). This is why software companies deliberately cripple basic versions---it's strategic product degradation to enable better price discrimination, not cost reduction.
	\end{solution}
\end{parts}
