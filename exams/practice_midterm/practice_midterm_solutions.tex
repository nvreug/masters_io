\subsection*{1. Short Answer Questions (30 points)}
\question
\begin{parts}
	\part[4] Profit-maximizing price:
	\begin{solution}
	$\frac{P - MC}{P} = \frac{1}{|\varepsilon|} = \frac{1}{4} \implies P - 15 = 0.25P \implies 0.75P = 15 \implies P^* = \$20$
	\end{solution}

	\part[4] Interpretation of $\xi_{jt}$:
	\begin{solution}
	$\xi_{jt}$ is unobserved product quality---factors affecting demand that the researcher doesn't observe (e.g., brand reputation, advertising, design).
	\end{solution}

	\part[4] Demographics fully eliminate IIA:
	\begin{solution}
	False. IIA still holds within demographic groups. Only mixed logit (random coefficients) fully relaxes IIA.
	\end{solution}

	\part[4] Cost shifters as instruments:
	\begin{solution}
	Relevance: Higher input costs $\rightarrow$ higher prices. Exogeneity: Input prices determined by upstream markets, not correlated with local demand shocks $\xi_{jt}$.
	\end{solution}

	\part[4] Cross-price elasticity depends on similarity:
	\begin{solution}
	False. In logit, $\eta_{jk} = -\alpha p_k s_k$---depends only on $k$'s price and share, not on similarity between $j$ and $k$. This is the IIA limitation.
	\end{solution}

	\part[3] MC pricing gives zero profit:
	\begin{solution}
	False. $\pi = pq - F - cq = cq - F - cq = -F < 0$. MC pricing requires government subsidies.
	\end{solution}

	\part[4] False CS formula:
	\begin{solution}
	Incorrect. Correct formula is $CS = \frac{1}{|\alpha|} \ln(1 + e^{\delta_1} + e^{\delta_2})$. The proposed formula is missing: (1) the log, (2) the $1/|\alpha|$ scaling, (3) the outside option term.
	\end{solution}

	\part[3] High-type IC always binds:
	\begin{solution}
	True. At optimum, high type is just indifferent between high and low product. If slack, firm could raise high-type price.
	\end{solution}
\end{parts}

\subsection*{2. Demand Estimation (30 points)}
\question
\begin{parts}
	\part[6] Define $\delta_{jt}$ and derive Berry inversion:
	\begin{solution}
	$\delta_{jt} = \beta \cdot \text{screen}_j + \alpha \cdot p_{jt} + \xi_{jt}$ (part of utility common to all consumers).

	Berry inversion: $\delta_{jt} = \ln(s_{jt}) - \ln(s_{0t})$
	\end{solution}

	\part[6] Regression equation:
	\begin{solution}
	$\ln(s_{jt}) - \ln(s_{0t}) = \beta \cdot \text{screen}_j + \alpha \cdot p_{jt} + \xi_{jt}$

	Dependent variable: $\ln(s_{jt}) - \ln(s_{0t})$. Regressors: $\text{screen}_j$, $p_{jt}$. Error: $\xi_{jt}$.
	\end{solution}

	\part[6] OLS bias:
	\begin{solution}
	$\text{Cov}(p_{jt}, \xi_{jt}) > 0$: high-quality phones command higher prices.

	Upward bias (less negative $\hat{\alpha}$): OLS attributes high demand for expensive phones to low price sensitivity rather than unobserved quality.
	\end{solution}

	\part[6] Cost shifters exclusion restriction:
	\begin{solution}
	If chip prices affect product quality (e.g., better chips $\rightarrow$ faster phones $\rightarrow$ higher $\xi$), then chip prices are correlated with $\xi_{jt}$, violating exclusion.
	\end{solution}

	\part[6] Advantage of demographic interactions:
	\begin{solution}
	Allows heterogeneous preferences across consumers, partially addressing IIA: older vs. younger consumers may prefer different screen sizes, generating more realistic substitution patterns.
	\end{solution}

\end{parts}

\subsection*{3. Two-Part Tariffs (20 points)}
\question
\begin{parts}
	\part[10] Profit as a function of $F$ and $p$:
	\begin{solution}
	$q = 30 - 3p$, choke price $p = 10$.

	$CS(p) = \frac{1}{2}(30-3p)(10-p) = \frac{3}{2}(10-p)^2$, so max fee $F = \frac{3}{2}(10-p)^2$.

	$\Pi = 200 \left[ \frac{3}{2}(10-p)^2 + (p-1)(30-3p) \right]$
	\end{solution}

	\part[10] Show $p^* = c = 1$:
	\begin{solution}
	Per-consumer profit: $\pi(p) = \frac{3}{2}(10-p)^2 + (p-1)(30-3p)$

	$\frac{d\pi}{dp} = -3(10-p) + (30-3p) - 3(p-1) = -30 + 3p + 30 - 3p - 3p + 3 = 3 - 3p$

	$3 - 3p = 0 \Rightarrow p^* = 1 = c$
	\end{solution}
\end{parts}

\subsection*{4. Self-Selection (20 points)}
\question
\begin{parts}
	\part[6] Perfect price discrimination:
	\begin{solution}
	Corporate: $P_B = \$800$. Leisure: $P_E = \$150$.

	Profit $= 50(800 - 50) + 100(150 - 50) = 50(750) + 100(100) = 37,500 + 10,000 = \$47,500$
	\end{solution}

	\part[6] IC constraint for corporate travelers:
	\begin{solution}
	$800 - P_B \geq 300 - P_E \implies P_B - P_E \leq 500$
	\end{solution}

	\part[8] Self-selection prices:
	\begin{solution}
	Binding: Leisure IR ($P_E = 150$) and Corporate IC ($P_B = P_E + 500 = 650$).

	Check leisure IC: $150 - 150 = 0 \geq 200 - 650 = -450$ \checkmark

	Profit $= 50(650 - 50) + 100(150 - 50) = 50(600) + 100(100) = 30,000 + 10,000 = \$40,000$
	\end{solution}
\end{parts}
