\begin{frame}{Recap: Logit, Berry inversion, and identification}
	\begin{wideitemize}
		\item Mean utility: $\delta_{jt} = x_{jt} \beta + \alpha p_{jt} + \xi_{jt}$ \quad ($\alpha < 0$)
		\item \textbf{Berry inversion:}
		\begin{align*}
			\ln(s_{jt}) - \ln(s_{0t}) = x_{jt} \beta + \alpha p_{jt} + \xi_{jt}
		\end{align*}
		\item \textbf{The identification problem:}
		\begin{wideitemize}
			\vspace{5pt}
			\item Firms observe $\xi_{jt}$ when setting prices
			\item $\Rightarrow$ $\text{Cov}(p_{jt}, \xi_{jt}) > 0$ $\Rightarrow$ OLS is biased
			\item Need supply shifters (IVs) to identify demand
		\end{wideitemize}
		\item \textbf{Today:} Direction of bias, IV strategies, and limitations
	\end{wideitemize}
\end{frame}

\begin{frame}{Plan for today}
  \begin{wideenumerate}

    \item \textbf{Direction of bias}
    \item Uber example
    \item Instrumental variables
    \item Elasticities and limitations

    \item Why demographics help
  \end{wideenumerate}
\end{frame}

\begin{frame}{Worked example: Bias direction}
	\begin{wideitemize}
		\item \textbf{Question:} You estimate a logit demand model using OLS and get $\hat{\alpha} = -0.3$. A colleague says the true $\alpha$ is likely $-0.5$.
		\item Is this consistent with endogeneity bias? Why?
		\pause
		\item \textbf{Answer:} Yes!
		\begin{wideitemize}
			\vspace{5pt}
			\item OLS overstates how much consumers like expensive products
			\item So OLS finds a smaller (less negative) price coefficient
			\item $-0.3 > -0.5$, so this is exactly what we'd expect
		\end{wideitemize}
	\end{wideitemize}
\end{frame}

\begin{frame}{Direction of bias}
	\begin{wideitemize}
		\item If high-quality products have high prices...
		\item OLS sees: high price, but demand still high (because of $\xi$)
		\item OLS concludes: price doesn't hurt demand much
		\item \textbf{Result:} $\hat{\alpha}$ biased toward zero (less negative than truth)
	\end{wideitemize}
\end{frame}

\begin{frame}{Case study: Uber surge pricing}
	\begin{wideitemize}
		\item Uber's surge pricing: prices rise when demand is high
		\item \textbf{Why is surge pricing endogenous?} \pause
		\item Surge responds to BOTH:
		\begin{wideitemize}
			\vspace{5pt}
			\item \textbf{Supply shocks:} Fewer drivers available $\rightarrow$ surge up
			\item \textbf{Demand shocks:} Concert ends, rain starts $\rightarrow$ surge up
		\end{wideitemize}
		\item Can't just regress rides on surge multiplier!
		\item Same problem as before: observing equilibrium prices
	\end{wideitemize}
\end{frame}

\begin{frame}{The solution: Uber's price ``wiggle'' experiment}
	\begin{wideitemize}
		\item Uber's insight: \textbf{randomize} prices!
		\item \textbf{Cohen et al. (2016):} Uber experimentally varied surge multipliers
		\item Same time, same location $\rightarrow$ different riders see different prices
		\item Key: Some riders randomly see 1.9x, others see 2.1x
		\item Demand conditions are identical, only price differs
	\end{wideitemize}
\end{frame}

\begin{frame}{Uber experiment: Results}
	\begin{wideitemize}
		\item Result: demand elasticity $\approx -0.5$
		\item \textbf{Inelastic!} A 10\% price increase $\rightarrow$ only 5\% fewer rides
		\item Why so inelastic?
		\begin{wideitemize}
			\vspace{5pt}
			\item When you need a ride, you \textit{need} a ride
			\item Limited substitutes late at night
		\end{wideitemize}
		\item \textbf{Key lesson:} Experiments give clean identification
		\item But most industries can't randomize prices...
	\end{wideitemize}
\end{frame}

\begin{frame}{Why experiments are powerful}
	\begin{wideitemize}
		\item No confounding: price variation is independent of demand shocks
		\item Clean identification of the demand curve
		\item \textbf{But:}
		\begin{wideitemize}
			\vspace{5pt}
			\item Tech companies can run experiments
			\item Traditional industries can't randomize prices
			\item Most IO settings require \textbf{instrumental variables}
		\end{wideitemize}
	\end{wideitemize}
\end{frame}

\begin{frame}{Instrumental variables: the solution}
	\begin{wideitemize}
		\item Need variables $z$ that are: \pause
		\begin{wideenumerate}
			\vspace{5pt}
			\item \textbf{Relevant:} Correlated with price ($\text{Cov}(z, p) \neq 0$)
			\item \textbf{Exogenous:} Uncorrelated with $\xi$ ($\text{Cov}(z, \xi) = 0$)
		\end{wideenumerate}
		\item These are cost shifters or other supply-side variables
	\end{wideitemize}
\end{frame}

\begin{frame}{Common IVs in demand estimation}
	Three main approaches used in the literature:
	\begin{wideenumerate}
		\item \textbf{Cost shifters:} Input prices, exchange rates
		\item \textbf{Hausman IVs:} Prices in other markets
		\item \textbf{BLP IVs:} Characteristics of competing products
	\end{wideenumerate}
	\vspace{1em}
	Let's examine each in detail...
\end{frame}

\begin{frame}{IV \#1: Cost shifters}
	\begin{wideitemize}
		\item \textbf{Idea:} Use variables that shift marginal costs
		\begin{wideitemize}
			\vspace{3pt}
			\item Input prices (steel, labor, energy)
			\item Exchange rates (for imported goods)
			\item Tariffs, taxes, regulations
		\end{wideitemize} \pause
		\item \textbf{Relevance:} Higher costs $\rightarrow$ higher prices
		\item \textbf{Exogeneity:} Cost shocks don't directly affect consumer utility
		\begin{wideitemize}
			\vspace{3pt}
			\item Steel prices don't make you like/dislike a car
		\end{wideitemize} \pause
		\item \textbf{Problem:} Cost data rarely observed!
		\begin{wideitemize}
			\vspace{3pt}
			\item Firms don't disclose detailed cost information
			\item Even aggregate input prices may be hard to match to products
		\end{wideitemize}
	\end{wideitemize}
\end{frame}

\begin{frame}{IV \#2: Hausman instruments}
	\begin{wideitemize}
		\item \textbf{Idea:} Use prices of the same product in \textit{other markets}
		\begin{wideitemize}
			\vspace{3pt}
			\item Coca-Cola price in Chicago as IV for Coca-Cola price in Phoenix
		\end{wideitemize} \pause
		\item \textbf{Why it works:}
		\begin{wideitemize}
			\vspace{3pt}
			\item Same product has similar costs across markets
			\item Cost shocks (ingredients, shipping) affect prices everywhere
		\end{wideitemize}
		\item \textbf{Key assumption:} Demand shocks $\xi_{jt}$ are independent across markets
		\begin{wideitemize}
			\vspace{3pt}
			\item Chicago's unobserved demand factors don't affect Phoenix
		\end{wideitemize}
	\end{wideitemize}
\end{frame}

\begin{frame}{Worked example: When do Hausman IVs fail?}
	\begin{wideitemize}
		\item \textbf{Question:} You want to use Pepsi prices in Denver as an IV for Pepsi prices in Phoenix. When might this fail?
		\pause
		\item \textbf{Answer:}
		\item Hausman IVs assume $\xi_{jt}$ is independent across markets
		\item \textbf{This fails when:}
		\begin{wideitemize}
			\vspace{3pt}
			\item National advertising campaigns affect all markets
			\item Nationwide product reformulation (``New Coke'')
			\item Coordinated promotions or seasonal demand patterns
		\end{wideitemize}
		\item \textbf{Relevance is OK:} Cost shocks do affect prices everywhere
		\item \textbf{Exogeneity is the problem:} Demand shocks may also be correlated!
	\end{wideitemize}
\end{frame}

\begin{frame}{IV \#3: BLP instruments}
	\begin{wideitemize}
		\item \textbf{Idea:} Use characteristics of \textit{other products} in the market
		\item \textbf{Key assumption:}
		\begin{align*}
			E(\xi_{jt} | x_{t}) = 0
		\end{align*}
		``Observed characteristics are mean independent of unobserved characteristics''
		\item \textbf{Common BLP instruments:}
		\begin{wideenumerate}
			\vspace{5pt}
			\item Own product characteristics (not price!)
			\item Sum of characteristics of \textit{own-firm} products
			\item Sum of characteristics of \textit{competitor} products
		\end{wideenumerate}
		\item \textbf{Intuition:} More/closer competitors $\rightarrow$ more competition $\rightarrow$ lower prices
	\end{wideitemize}
\end{frame}

\begin{frame}{BLP instruments: Example}
	\begin{wideitemize}
		\item Consider estimating demand for the Honda Civic
		\item \textbf{BLP instruments could include:}
		\begin{wideitemize}
			\vspace{3pt}
			\item Civic's own characteristics: horsepower, MPG, size
			\item Sum of Honda's other cars' characteristics (Accord, CR-V)
			\item Sum of competitors' characteristics (Corolla, Mazda3, etc.)
		\end{wideitemize} \pause
		\item \textbf{Why exogenous?}
		\begin{wideitemize}
			\vspace{3pt}
			\item Toyota's decision about Corolla's horsepower doesn't affect Civic's unobserved quality $\xi$
		\end{wideitemize}
		\item \textbf{Why relevant?}
		\begin{wideitemize}
			\vspace{3pt}
			\item If Toyota improves Corolla, Honda faces more competition
			\item This pressures Honda to lower Civic's price
		\end{wideitemize}
	\end{wideitemize}
\end{frame}

\begin{frame}{IV \#4: Fan/Waldfogel instruments}
	\begin{wideitemize}
		\item \textbf{Idea:} Use demographics in \textit{other markets} where the product is sold
		\item Example: Fan (2013) - Newspaper demand
		\begin{wideitemize}
			\vspace{3pt}
			\item Newspapers sold in multiple counties
			\item Use demographics in other counties as IVs for price
		\end{wideitemize} \pause
		\item \textbf{Why it works:}
		\begin{wideitemize}
			\vspace{3pt}
			\item Product characteristics shaped by demographics in ALL markets where sold
			\item Demographics elsewhere affect costs/design but not local demand
		\end{wideitemize}
		\item \textbf{Validity:} $\xi_{jt}$ not correlated across markets (same as Hausman)
		\item \textbf{Concern:} Which markets a firm enters may be endogenous
	\end{wideitemize}
\end{frame}

\begin{frame}{Worked example: Evaluating an IV}
	\begin{wideitemize}
		\item \textbf{Question:} A researcher proposes using gasoline prices as an IV for car prices. Is this valid?
		\pause
		\item \textbf{Relevance:} Higher gas prices $\rightarrow$ higher operating costs $\rightarrow$ might affect car prices? Maybe weakly.
		\item \textbf{Exogeneity:} Do gas prices affect car quality $\xi_{jt}$?
		\begin{wideitemize}
			\vspace{3pt}
			\item Gas prices affect \textit{demand} for fuel-efficient cars
			\item This might shift which cars look ``good'' to consumers
			\item Potentially problematic!
		\end{wideitemize}
		\item \textbf{Overall:} Probably not a great IV
	\end{wideitemize}
\end{frame}

\begin{frame}{Summary: IV conditions}
	\begin{wideitemize}
		\item For $z$ to be a valid IV:
		\begin{wideenumerate}
			\vspace{5pt}
			\item \textbf{Relevant:} $z$ must predict prices
			\begin{wideitemize}
				\item Can test this! Run first-stage regression
			\end{wideitemize}
			\item \textbf{Exogenous:} $z$ must not affect demand directly
			\begin{wideitemize}
				\item Cannot test this directly (requires economic reasoning)
			\end{wideitemize}
		\end{wideenumerate}
		\item This is the standard IV framework from econometrics
		\item Applied to demand estimation: use supply-side variation
	\end{wideitemize}
\end{frame}

\begin{frame}{What makes a good IV? Summary}
	\begin{center}
	\begin{tabular}{lcc}
		\toprule
		\textbf{Instrument} & \textbf{Relevance} & \textbf{Exogeneity} \\
		\midrule
		Cost shifters (input prices) & Strong & Usually OK \\
		Hausman IVs (prices elsewhere) & Strong & Debatable \\
		BLP IVs (competitor chars) & Medium & Usually OK \\
		Fan/Waldfogel (demographics elsewhere) & Medium & Debatable \\
		Experiments (Uber wiggle) & Perfect & Perfect \\
		\bottomrule
	\end{tabular}
	\end{center}
	\vspace{10pt}
	\begin{wideitemize}
		\item No perfect instrument in observational data
		\item \textbf{Best practice:} Use multiple IVs, check robustness
		\item \textbf{Experiments} are gold standard but often infeasible
	\end{wideitemize}
\end{frame}

% ELASTICITIES AND LIMITATIONS
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{frame}{From estimation to elasticities}
	\begin{wideitemize}
		\item We've seen how to estimate demand with logit
		\item \textbf{Why do we care?}
		\item Demand estimates let us compute \textbf{elasticities}
		\begin{wideitemize}
			\vspace{3pt}
			\item How sensitive are consumers to price changes?
			\item How do consumers substitute between products?
		\end{wideitemize}
		\item These are crucial for policy: mergers, pricing, welfare
		\item \textbf{But:} the logit model has important limitations...
	\end{wideitemize}
\end{frame}

\begin{frame}{Logit elasticities}
	\begin{wideitemize}
		\item Given shares $s_{jt} = \frac{\exp(\delta_{jt})}{1 + \sum_k \exp(\delta_{kt})}$
		\item We can derive price elasticities:
		\begin{align*}
			\eta_{jj} &= \frac{\partial s_{jt}}{\partial p_{jt}} \frac{p_{jt}}{s_{jt}} = \alpha p_{jt} (1 - s_{jt}) \quad \text{(own-price)} \\[10pt]
			\eta_{jk} &= \frac{\partial s_{jt}}{\partial p_{kt}} \frac{p_{kt}}{s_{jt}} = -\alpha p_{kt} s_{kt} \quad \text{(cross-price)}
		\end{align*}
		\item Note: $\alpha < 0$, so own-price elasticity is \textbf{negative} (as expected)
	\end{wideitemize}
\end{frame}

\begin{frame}{Limitation \#1: Own-price elasticity}
	\begin{wideitemize}
		\item Own-price elasticity: $\eta_{jj} = \alpha p_{jt} (1 - s_{jt})$
		\item With many products: $(1 - s_{jt}) \approx 1$
		\item So: $\eta_{jj} \approx \alpha p_{jt}$
		\item \textbf{Implication:} Own-price elasticity is proportional to price! \pause
		\item Lower-priced products have more \textit{inelastic} demand
		\item Higher-priced products have more \textit{elastic} demand
		\item \textbf{Question:} Is this realistic?
		\begin{wideitemize}
			\vspace{3pt}
			\item Generic cereal (low price) should be more elastic than organic cereal?
			\item Budget airline (low price) more inelastic than first class?
		\end{wideitemize}
	\end{wideitemize}
\end{frame}

\begin{frame}{Limitation \#2: Unrealistic substitution patterns}
	\begin{wideitemize}
		\item Cross-price elasticity: $\eta_{jk} = -\alpha p_{kt} s_{kt}$
		\item This doesn't depend on product $j$ at all!
		\item \textbf{Implication:} All products have the \textit{same} cross-elasticity with product $k$ \pause
		\item Example: BMW raises its price
		\begin{wideitemize}
			\vspace{3pt}
			\item Logit says: same fraction go to Mercedes as to Honda Civic!
			\item Reality: BMW consumers mostly switch to Mercedes, Audi
		\end{wideitemize}
		\item Substitution depends on \textbf{market share}, not product similarity
	\end{wideitemize}
\end{frame}

\begin{frame}{Diversion ratios in logit}
	\begin{wideitemize}
		\item When product $k$'s price rises, where do consumers go?
		\item \textbf{Diversion ratio} from $k$ to $j$:
		\begin{align*}
			D_{k \to j} = \frac{\partial s_{jt} / \partial p_{kt}}{\partial s_{kt} / \partial p_{kt}} = \frac{s_{jt}}{1 - s_{kt}}
		\end{align*}
		\item Substitution is proportional to \textbf{market share}, not similarity! \pause
		\item \textbf{Why this matters:}
		\begin{wideitemize}
			\vspace{3pt}
			\item Merger analysis: which products are close substitutes?
			\item Product positioning: who are my real competitors?
			\item Logit can't answer these questions well
		\end{wideitemize}
	\end{wideitemize}
\end{frame}

\begin{frame}{Why logit has unrealistic substitution}
	\begin{wideitemize}
		\item The problem comes from the \textbf{logit error structure}
		\item Key assumption: $\varepsilon_{ijt}$ are \textit{independent} across products
		\item This means: your taste shock for BMW is unrelated to your taste shock for Mercedes
		\item The model doesn't ``know'' that BMW and Mercedes are similar \pause
		\item \textbf{Solution:} Allow for consumer heterogeneity
		\begin{wideitemize}
			\vspace{3pt}
			\item High-income consumers prefer both BMW and Mercedes
			\item When BMW's price rises, these consumers switch to Mercedes
		\end{wideitemize}
	\end{wideitemize}
\end{frame}

\begin{frame}{The demographic interaction model}
	\begin{wideitemize}
		\item We estimate a version of \textbf{Berry, Levinsohn \& Pakes (1995)} --- ``BLP''
		\item Allow preferences to vary across consumers:
		\begin{align*}
			u_{ijt} = x_{jt} \beta_i + \alpha_i p_{jt} + \xi_{jt} + \varepsilon_{ijt}
		\end{align*}
		\item $\beta_i$, $\alpha_i$: consumer $i$'s taste for characteristics and price
		\item \textbf{Key question:} What drives heterogeneity in $\beta_i$ and $\alpha_i$?
		\item \textbf{Our approach:} Observed demographics
		\begin{align*}
			\beta_{ik} = \beta_k + \sum_{\ell=1}^L \pi_{k\ell} D_{i\ell}, \qquad \alpha_i = \alpha + \sum_{\ell=1}^L \pi_{\alpha\ell} D_{i\ell}
		\end{align*}
		\item $D_{i\ell}$: demographic $\ell$ for consumer $i$ (income, age, family size, etc.)
	\end{wideitemize}
\end{frame}

\begin{frame}{Rewriting in terms of mean utility}
	\begin{wideitemize}
		\item Substituting the expressions for $\beta_{ik}$ and $\alpha_i$:
		\begin{align*}
			u_{ijt} = \underbrace{x_{jt} \beta + \alpha p_{jt} + \xi_{jt}}_{\delta_{jt} \text{ (mean utility)}} + \underbrace{\sum_{k,\ell} \pi_{k\ell} D_{i\ell} x_{jt,k} + \sum_{\ell} \pi_{\alpha\ell} D_{i\ell} p_{jt}}_{\mu_{ijt}} + \varepsilon_{ijt}
		\end{align*}
		\item \textbf{Mean utility} $\delta_{jt}$: part common to all consumers
		\begin{wideitemize}
			\item Contains ``\textbf{linear parameters}'' $(\beta, \alpha)$ --- enter linearly in $\delta$
		\end{wideitemize}
		\item \textbf{Individual deviation} $\mu_{ijt}$: how consumer $i$ differs from average
		\begin{wideitemize}
			\item Contains ``\textbf{nonlinear parameters}'' $(\pi_{k\ell})$ --- enter nonlinearly via shares
		\end{wideitemize}
		\item Note: $\alpha < 0$ (higher price $\rightarrow$ lower utility)
	\end{wideitemize}
\end{frame}

\begin{frame}{From utility to demand}
	\begin{wideitemize}
		\item \textbf{Individual choice probability} (with T1EV errors):
		\begin{align*}
			s_{ijt} = \frac{\exp(\delta_{jt} + \mu_{ijt})}{1 + \sum_{k=1}^J \exp(\delta_{kt} + \mu_{ikt})}
		\end{align*}
		\item \textbf{Market share} (aggregate over consumers):
		\begin{align*}
			s_{jt} = \frac{1}{I} \sum_{i=1}^{I} s_{ijt}
		\end{align*}
		\item This is \textbf{demand}: the fraction of consumers who buy product $j$
		\item Key: shares depend on mean utilities $\delta$ and demographics $D_i$
	\end{wideitemize}
\end{frame}

\begin{frame}{Elasticities with demographic interactions}
	\begin{wideitemize}
		\item \textbf{Elasticities with heterogeneous consumers:}
		\[  \eta_{jkt} =  \left\{
		\begin{array}{ll}
		\frac{p_{jt}}{s_{jt}} \sum_i \alpha_i s_{ijt} (1-s_{ijt}) & \text{if } j=k \\[8pt]
		-\frac{p_{kt}}{s_{jt}} \sum_i \alpha_i s_{ijt} s_{ikt} & \text{otherwise} \\
		\end{array}
		\right. \]
		\item $s_{ijt}$: probability that consumer $i$ purchases product $j$
		\item \textbf{Key insight:} Cross-elasticity now depends on $\sum_i s_{ijt} s_{ikt}$
		\begin{wideitemize}
			\item Products that attract \textit{similar consumers} have higher cross-elasticity
			\item BMW and Mercedes attract similar (high-income) consumers $\Rightarrow$ closer substitutes
		\end{wideitemize}
		\item Correlation in $\mu_{ijt}$ and $\mu_{ikt}$ drives realistic substitution patterns
	\end{wideitemize}
\end{frame}

\begin{frame}{Limitations: Demographics vs Mixed Logit}
	\begin{wideitemize}
		\item \textbf{Demographics model} (what we use):
		\begin{wideitemize}
			\vspace{5pt}
			\item Heterogeneity only from observed characteristics
			\item Substitution still restrictive within demographic groups
			\item Simple and tractable
		\end{wideitemize}
		\item \textbf{Mixed logit / random coefficients} (advanced):
		\begin{wideitemize}
			\vspace{5pt}
			\item Heterogeneity from unobserved factors too
			\item Fully flexible substitution patterns
			\item Computationally intensive (requires simulation)
		\end{wideitemize}
		\item For this course: demographics are sufficient
		\item Know that mixed logit exists for research
	\end{wideitemize}
\end{frame}

\begin{frame}{Key Points}
	\vspace{11pt}
	\begin{wideenumerate}
		\item \textbf{Price is endogenous}: Firms observe $\xi_{jt}$ when pricing $\rightarrow$ OLS biased toward zero
		\item \textbf{Solution}: Instrumental variables (cost shifters, BLP IVs, Hausman IVs)
		\item \textbf{Logit limitation}: Cross-elasticity doesn't depend on $j$ $\Rightarrow$ unrealistic substitution
		\item \textbf{Demographic model}: $u_{ijt} = \delta_{jt} + \sum_{k,\ell} \pi_{k\ell} D_{i\ell} x_{jt,k} + \varepsilon_{ijt}$
		\item Demographics $\Rightarrow$ heterogeneous preferences $\Rightarrow$ realistic substitution
	\end{wideenumerate}
\end{frame}
