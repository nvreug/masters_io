% CONSUMER SURPLUS
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{frame}{Why do we care about consumer surplus?}
	\begin{wideitemize}
		\item Policy analysis requires measuring welfare
		\item Questions we want to answer:
		\begin{wideitemize}
			\vspace{5pt}
			\item How much do consumers gain from a new product?
			\item How much do consumers lose from a merger?
			\item What is the welfare cost of a price increase?
		\end{wideitemize}
		\item Need a way to compute consumer surplus from our demand model
	\end{wideitemize}
\end{frame}

\begin{frame}{Consumer surplus in logit: the log-sum formula}
	\begin{wideitemize}
		\item For consumer $i$ in market $t$, expected utility from choosing among $J$ products:
		\begin{align*}
			E[\max_j u_{ijt}] = \ln\left[\sum_{j=0}^J \exp(\delta_{jt} + \mu_{ijt})\right] + \text{constant}
		\end{align*}
		\item This is the ``log-sum'' or ``inclusive value''
		\item Consumer surplus (in dollars):
		\begin{align*}
			CS_{it} = \frac{1}{|\alpha_i|} \ln\left[\sum_{j=0}^J \exp(\delta_{jt} + \mu_{ijt})\right]
		\end{align*}
		\item Divide by $|\alpha_i|$ (consumer $i$'s price coefficient) to convert to dollars
	\end{wideitemize}
\end{frame}

\begin{frame}{Intuition for the log-sum}
	\begin{wideitemize}
		\item Think of choosing the best option as a lottery
		\item Each product gives you a random utility draw
		\item \textbf{Expected value of the BEST draw} is the log-sum
		\item Key insights:
		\begin{wideitemize}
			\vspace{5pt}
			\item More options $\rightarrow$ higher CS (more lottery tickets)
			\item Better options $\rightarrow$ higher CS (higher $\delta_j$)
			\item Higher price sensitivity $\rightarrow$ divide by larger $|\alpha|$
		\end{wideitemize}
	\end{wideitemize}
\end{frame}

\begin{frame}{Worked example: CS change from removing a product}
	\begin{wideitemize}
		\item \textbf{Question:} Market has 2 products with $\delta_1 = 1$, $\delta_2 = 0.5$. Outside option has $\delta_0 = 0$. Suppose $\alpha = -0.5$.
		\item What is the consumer surplus loss if product 1 is removed?
	\end{wideitemize}
	\vspace{15pt}
	\centering
	\textit{Take 3 minutes to solve this.}
\end{frame}

\begin{frame}{Worked example: CS change (solution)}
	\begin{wideitemize}
		\item \textbf{Solution:}
		\item \textbf{Before removal:}
		\begin{align*}
			CS^{\text{before}} = \frac{1}{|\alpha|} \ln(e^0 + e^1 + e^{0.5}) = 2 \ln(1 + 2.72 + 1.65) = 2 \ln(5.37) = 3.36
		\end{align*}
		\item \textbf{After removal:}
		\begin{align*}
			CS^{\text{after}} = \frac{1}{|\alpha|} \ln(e^0 + e^{0.5}) = 2 \ln(1 + 1.65) = 2 \ln(2.65) = 1.95
		\end{align*}
		\item \textbf{Loss:} $3.36 - 1.95 = 1.41$ dollars per consumer
	\end{wideitemize}
\end{frame}

\begin{frame}{Welfare from new products: Red Bus / Blue Bus}
	\begin{wideitemize}
		\item The log-sum formula lets us value welfare from new products
		\item But there's a problem: the \textbf{Red Bus / Blue Bus} example
		\item \textbf{Setup:} Consumers choose how to commute
		\begin{wideitemize}
			\vspace{5pt}
			\item Choices: Car, Red Bus
			\item Half choose Car, half choose Red Bus
			\item So: $\delta_{\text{car}} = \delta_{\text{red bus}} = 0$
		\end{wideitemize}
	\end{wideitemize}
\end{frame}

\begin{frame}{Red Bus / Blue Bus: introducing a new option}
	\begin{wideitemize}
		\item Artificially introduce a \textbf{Blue Bus}
		\item But consumers are \textbf{color-blind}!
		\item Blue Bus is identical to Red Bus in every way
		\item Now: half choose Car, rest split between buses
		\item \textbf{Reality:} Welfare should NOT change
		\begin{wideitemize}
			\vspace{5pt}
			\item It's the same bus, just different color
			\item No real new option for consumers
		\end{wideitemize}
	\end{wideitemize}
\end{frame}

\begin{frame}{What does logit predict?}
	\begin{wideitemize}
		\item Suppose we only observe data \textbf{pre-introduction} of Blue Bus
		\item \textbf{Pre-introduction:}
		\begin{itemize}
			\item Normalize: $\delta_{\text{car}} = 0$. Then $\delta_{\text{red bus}} = 0$ (since equal shares)
			\item Inclusive value: $\ln(e^0 + e^0) = \ln(2)$
		\end{itemize}
		\item \textbf{Post-introduction} (using logit to predict):
		\begin{itemize}
			\item $\delta_{\text{blue bus}} = \delta_{\text{red bus}} = 0$ (same bus)
			\item Logit predicts: $s_{\text{car}} = s_{\text{blue}} = s_{\text{red}} = 1/3$
			\item Inclusive value: $\ln(e^0 + e^0 + e^0) = \ln(3)$
		\end{itemize}
		\item Logit says welfare \textbf{increased}! But nothing real changed...
	\end{wideitemize}
\end{frame}

\begin{frame}{Red Bus / Blue Bus: what went wrong?}
	\begin{wideitemize}
		\item \textbf{Main issue:} We get an extra logit draw when introducing Blue Bus
		\item Logit gives an extra ``lottery ticket'' for each new product
		\item It doesn't know that buses are close substitutes
	\end{wideitemize}
\end{frame}

\begin{frame}{Solution: use observed post-introduction shares}
	\begin{wideitemize}
		\item \textbf{Key insight:} If we observe the market \textit{after} introducing Blue Bus, we can correct this
		\item \textbf{Observed shares:} $s_{\text{car}} = 0.5$, $s_{\text{blue}} = s_{\text{red}} = 0.25$
		\item Back out $\delta$ from observed shares:
		\begin{align*}
			\delta_{\text{blue bus}} = \delta_{\text{red bus}} = \ln(0.25) - \ln(0.5) = \ln(0.5)
		\end{align*}
		\item \textbf{Corrected inclusive value:}
		\begin{align*}
			\ln(e^0 + 2 \times e^{\ln(0.5)}) = \ln(1 + 2 \times 0.5) = \ln(2)
		\end{align*}
		\item This is the \textbf{correct answer} --- no welfare change!
	\end{wideitemize}
\end{frame}

\begin{frame}{Key takeaways: welfare from new products}
	\begin{wideitemize}
		\item \textbf{Problem:} Introducing new products adds extra logit draws, biasing welfare up
		\item \textbf{Solution:} Observing post-introduction market shares can correct for this
		\begin{wideitemize}
			\vspace{5pt}
			\item Fitting the shares exactly means the extra logit gain is offset by lower $\delta$'s
		\end{wideitemize}
		\item \textbf{Bottom line:} Be careful computing welfare from hypothetical new products
		\begin{wideitemize}
			\vspace{5pt}
			\item If you observe the new product in data, fit the shares first
			\item If purely counterfactual, results may be biased
		\end{wideitemize}
	\end{wideitemize}
\end{frame}

\begin{frame}{Petrin (2002): Counterfactual}

\begin{frame}{Monopoly pricing: setup}
\begin{wideitemize}
	\item \textbf{Goal:} Derive optimal pricing rule for a monopolist
	\item \underline{Setup}:
	\begin{wideitemize}
		\vspace{5pt}
		\item Inverse demand curve: $p = P(q)$
		\item Total cost: $C(q)$
		\item Monopolist chooses $q$ to maximize profit
	\end{wideitemize}
	\item \textbf{Profit function:}
	\begin{align*}
		\pi(q) = \underbrace{P(q) \cdot q}_{TR} - \underbrace{C(q)}_{TC}
	\end{align*}
\end{wideitemize}
\end{frame}

\begin{frame}{Deriving MR = MC}
	\begin{wideitemize}
		\item Take first-order condition: $\frac{d\pi}{dq} = 0$
		\begin{align*}
			\frac{d}{dq}\left[ P(q) \cdot q - C(q) \right] = 0
		\end{align*}
		\item Applying the product rule to $P(q) \cdot q$:
		\begin{align*}
			P(q) + P'(q) \cdot q - C'(q) = 0
		\end{align*}
		\item Rearranging:
		\begin{align*}
			\underbrace{P(q) + P'(q) \cdot q}_{MR} = \underbrace{C'(q)}_{MC}
		\end{align*}
		\item This gives us: $\boxed{MR = MC}$
	\end{wideitemize}
\end{frame}

\begin{frame}{Deriving the Lerner Index}
	\begin{wideitemize}
		\item Start from $MR = MC$: \quad $p + P'(q) \cdot q = MC$
		\item Rearrange:
		\begin{align*}
			p - MC = -P'(q) \cdot q
		\end{align*}
		\item Divide both sides by $p$:
		\begin{align*}
			\frac{p - MC}{p} = -P'(q) \cdot \frac{q}{p}
		\end{align*}
		\item Recall elasticity: $\varepsilon = \frac{dq}{dp} \cdot \frac{p}{q}$, so $\frac{1}{\varepsilon} = \frac{dp}{dq} \cdot \frac{q}{p} = P'(q) \cdot \frac{q}{p}$
		\item Therefore:
		\begin{align*}
			\boxed{\frac{p - MC}{p} = \frac{1}{|\varepsilon|}}
		\end{align*}
		\item This is the \textbf{Lerner Index}: markup equals inverse elasticity
	\end{wideitemize}
\end{frame}

\begin{frame}{The Lerner Index}
	\begin{wideitemize}
		\item The \textbf{Lerner Index} measures market power:
		\begin{align*}
			L = \frac{p - MC}{p} = \frac{1}{|\varepsilon|}
		\end{align*}
		\item Equivalently, solving for optimal price:
		\begin{align*}
			p = \frac{MC}{1 + \frac{1}{\varepsilon}} = \frac{MC}{1 - \frac{1}{|\varepsilon|}}
		\end{align*}
		\item \textbf{Interpretation:}
		\begin{wideitemize}
			\vspace{5pt}
			\item $L = 0$: perfect competition (price = MC)
			\item $L$ close to 1: high market power
			\item More elastic demand $\rightarrow$ lower markup
		\end{wideitemize}
	\end{wideitemize}
\end{frame}

\begin{frame}{Using estimated elasticities for pricing}
	\begin{wideitemize}
		\item Suppose you work as an economist in a firm pricing a product.
		\item You know MC. You \textbf{estimate} the demand elasticity $\varepsilon$.
		\item Then just plug into the elasticity rule:
		\begin{align*}
			p = \frac{MC}{1 + \frac{1}{\varepsilon}}
		\end{align*}
		\item \textbf{This is why demand estimation matters!}
		\begin{wideitemize}
			\vspace{5pt}
			\item Estimate demand $\rightarrow$ get elasticities $\rightarrow$ compute optimal prices
		\end{wideitemize}
	\end{wideitemize}
\end{frame}

\begin{frame}{Regulating monopolies}
	\begin{wideitemize}
		\item How can we correct the market failure of monopoly?
		\item \textbf{Option 1:} Break up the monopoly (antitrust)
		\begin{wideitemize}
			\vspace{5pt}
			\item Standard Oil (1911), AT\&T (1984)
			\item Current debates: Google, Meta, Amazon
		\end{wideitemize}
		\item Sometimes breaking up isn't possible (natural monopolies)
		\begin{wideitemize}
			\vspace{5pt}
			\item Power plants, bridges, water utilities
			\item High fixed costs make one firm efficient
		\end{wideitemize}
		\item How should we regulate these natural monopolies?
	\end{wideitemize}
\end{frame}

\begin{frame}{Marginal cost pricing}
	\begin{wideitemize}
		\item \textbf{Idea:} Force the monopolist to set $p=MC$
		\begin{wideitemize}
			\vspace{5pt}
			\item This is the competitive price; no deadweight loss
		\end{wideitemize}
		\item \textbf{Problem:} Suppose cost is $C(q) = F + cq$ (fixed cost + marginal cost)
		\item If $p=MC=c$, then profit is:
		\begin{align*}
			\pi = pq - F - cq = cq - F - cq = -F < 0
		\end{align*}
		\item Firm makes \textbf{negative} profit and would shut down!
		\item Solution: government \textbf{subsidy} of $F$
	\end{wideitemize}
\end{frame}

\begin{frame}{Average cost pricing}
	\begin{wideitemize}
		\item \textbf{Alternative:} Force the monopolist to set $p=AC$
		\item This is the price where the monopolist makes zero profit:
		\begin{wideitemize}
			\vspace{5pt}
			\item If $\pi = 0$ then $TR = TC$
			\item So $pq = TC$, which means $p = TC/q = AC$
		\end{wideitemize}
		\item The monopolist can only exercise market power to cover fixed costs
		\item No government subsidy needed
		\item Commonly used for utilities (``rate of return regulation'')
	\end{wideitemize}
\end{frame}

\begin{frame}{Price caps}
	\begin{wideitemize}
		\item \textbf{Price cap regulation:} Set a maximum price the firm can charge
		\item Often indexed to inflation (``CPI-X'' regulation)
		\begin{wideitemize}
			\vspace{5pt}
			\item Price can rise with inflation, minus an efficiency factor X
		\end{wideitemize}
		\item Advantage: incentivizes cost reduction
		\begin{wideitemize}
			\vspace{5pt}
			\item Under AC pricing, firm has no incentive to cut costs
			\item Under price caps, cost savings become profit
		\end{wideitemize}
		\item Common in UK utilities, telecoms
	\end{wideitemize}
\end{frame}

\begin{frame}{Key Points}
	\vspace{11pt}
	\begin{wideenumerate}
		\item \textbf{Log-sum formula}: $CS_{it} = \frac{1}{|\alpha_i|}\ln\left[\sum_j \exp(\delta_{jt} + \mu_{ijt})\right]$
		\item \textbf{Red Bus / Blue Bus}: Logit overcounts welfare from similar new products
		\item \textbf{Solution}: Use observed post-introduction shares to correct bias
		\item \textbf{Lerner Index}: $\frac{p - MC}{p} = \frac{1}{|\varepsilon|}$ measures market power
		\item \textbf{Monopoly regulation}: MC pricing (needs subsidy), AC pricing (zero profit), price caps
		\item More elastic demand $\rightarrow$ lower markup
	\end{wideenumerate}
\end{frame}
