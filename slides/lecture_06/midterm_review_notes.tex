\section{Demand Estimation}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\subsection{Key Concepts}
\begin{itemize}
	\item \textbf{Utility:} $u_{ijt} = \delta_{jt} + \varepsilon_{ijt}$ where $\delta_{jt} = x_{jt}\beta + \alpha p_{jt} + \xi_{jt}$ \quad ($\alpha < 0$)
	\item \textbf{Shares:} $s_{jt} = \frac{\exp(\delta_{jt})}{1 + \sum_{k} \exp(\delta_{kt})}$, \quad $s_{0t} = \frac{1}{1 + \sum_{k} \exp(\delta_{kt})}$
	\item \textbf{Price endogeneity:} High $\xi_{jt} \rightarrow$ high price $\Rightarrow$ OLS biases $\alpha$ toward zero (less negative)
	\item \textbf{IVs:} Cost shifters, Hausman (prices elsewhere), BLP (rival characteristics)
	\item \textbf{IIA problem:} Cross-elasticity $\eta_{jk}$ doesn't depend on $j$ $\Rightarrow$ unrealistic substitution
	\item \textbf{Demographics help:} $u_{ijt} = \delta_{jt} + \sum_{k,\ell} \pi_{k\ell} D_{i\ell} x_{jt,k} + \varepsilon_{ijt}$ $\Rightarrow$ richer substitution
\end{itemize}

\subsection{Cookbook: Berry Inversion}
\begin{enumerate}
	\item Compute $\delta_j = \ln(s_j) - \ln(s_0)$ for each product
	\item Interpretation: higher $\delta$ = more preferred (relative to outside option)
\end{enumerate}

\subsection{Cookbook: Elasticity Calculation}
\begin{enumerate}
	\item Own-price: $\eta_{jj} = \alpha p_j (1 - s_j)$. Since $\alpha < 0$, this is negative.
	\item Cross-price: $\eta_{jk} = -\alpha p_k s_k$. Since $\alpha < 0$, this is positive (substitutes).
	\item With small shares: $\eta_{jj} \approx \alpha p_j$ (higher price $\Rightarrow$ more elastic)
\end{enumerate}

\subsection{Cookbook: Consumer Surplus Change}
\begin{enumerate}
	\item Before: $CS^{\text{before}} = \frac{1}{|\alpha|} \ln(e^{\delta_0} + e^{\delta_1} + \ldots)$ where $\delta_0 = 0$
	\item After: $CS^{\text{after}} = \frac{1}{|\alpha|} \ln(\text{sum over remaining products})$
	\item Loss: $\Delta CS = CS^{\text{before}} - CS^{\text{after}}$
\end{enumerate}
\textbf{Common mistake:} Using $\sum \exp(\delta_j)$ instead of $\ln[\sum \exp(\delta_j)]$

\subsection{Demographic Interactions Model}
\begin{itemize}
	\item \textbf{Model:} $u_{ijt} = \delta_{jt} + \mu_{ijt} + \varepsilon_{ijt}$
	\item Mean utility: $\delta_{jt} = x_{jt}\beta + \alpha p_{jt} + \xi_{jt}$ (common to all consumers)
	\item Individual deviation: $\mu_{ijt} = \sum_{k,\ell} \pi_{k\ell} D_{i\ell} x_{jt,k}$ (e.g., income $\times$ price)
	\item \textbf{Advantage:} Richer substitution patterns---products that attract similar consumers are closer substitutes
	\item \textbf{Example:} If $\pi_{\text{inc,price}} > 0$, high-income consumers are less price-sensitive
\end{itemize}

\subsection{Red Bus / Blue Bus Problem}
\begin{itemize}
	\item \textbf{Setup:} Car and Red Bus, equal shares. Introduce identical Blue Bus.
	\item \textbf{Logit predicts:} CS increases (adds an extra logit draw)
	\item \textbf{Reality:} No real welfare gain (it's the same bus!)
	\item \textbf{Fix:} Use observed post-introduction shares to back out $\delta$'s. The lower shares for each bus offset the extra logit gain.
\end{itemize}

\subsection{Applications}
\begin{itemize}
	\item \textbf{Petrin (2002):} Valued minivan introduction at \$2.8B. Key insight: logit overstates CS gains; micro-data helps.
	\item \textbf{Nevo (2001):} Cereal demand. Demographics $\rightarrow$ richer substitution. Used to study market power.
\end{itemize}

\section{Pricing and Price Discrimination}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\subsection{Monopoly Pricing}
\begin{itemize}
	\item \textbf{Lerner index:} $L = \frac{p - MC}{p} = \frac{1}{|\varepsilon|}$ (markup = inverse elasticity)
	\item \textbf{MC pricing:} $p = MC$ $\Rightarrow$ DWL = 0, but firm needs subsidy if $F > 0$
	\item \textbf{AC pricing:} $p = AC$ $\Rightarrow$ zero profit, but DWL $> 0$ (some consumers with WTP $\in (MC, AC)$ don't buy)
\end{itemize}

\subsection{Cookbook: Selection by Indicators}
\begin{enumerate}
	\item Apply Lerner in each market: $\frac{p_i - MC}{p_i} = \frac{1}{|\varepsilon_i|}$
	\item Solve: $p_i = \frac{MC}{1 - 1/|\varepsilon_i|}$
	\item Result: higher price in more inelastic market
\end{enumerate}

\subsection{Cookbook: Two-Part Tariffs (Homogeneous Consumers)}
\begin{enumerate}
	\item Set $p^* = MC$ (maximizes total surplus)
	\item Compute quantity: $q^* = D(p^*)$
	\item Compute $CS = \frac{1}{2} \times q^* \times (\text{choke price} - p^*)$ for linear demand
	\item Set $F^* = CS$ (extract all surplus via fee)
	\item Profit = $F^*$ (margin on sales is zero)
\end{enumerate}

\subsection{Cookbook: Two-Part Tariffs (Heterogeneous Consumers)}
\begin{enumerate}
	\item At $p = 0$, compute $CS_H$ and $CS_L$ for each type
	\item Strategy A: Set $F = CS_H$ $\Rightarrow$ only H joins, profit $= F \times N_H$
	\item Strategy B: Set $F = CS_L$ $\Rightarrow$ both join, profit $= F \times (N_H + N_L)$
	\item Compare profits to choose optimal $F$
\end{enumerate}

\subsection{Cookbook: Self-Selection Menu Design}
\textbf{Setup:} Two types (H, L), two versions (Full, Stripped), WTP table given.
\begin{enumerate}
	\item Set IR for L to bind: $p_S = v_L^S$ (L gets zero surplus)
	\item Set IC for H to bind: $v_H^F - p_F = v_H^S - p_S$ $\Rightarrow$ $p_F = v_H^F - v_H^S + p_S$
	\item Verify IC for L: $v_L^S - p_S \geq v_L^F - p_F$ (should hold by construction)
	\item Profit $= (p_F - MC) \times N_H + (p_S - MC) \times N_L$
\end{enumerate}
\textbf{Key insight:} H gets ``information rent'' $= v_H^S - v_L^S$; L gets zero surplus.

\subsection{Cookbook: Checking a Given Menu}
\begin{enumerate}
	\item For each type, compute $CS = WTP - price$ for each option
	\item Each type buys the option with highest CS (if $\geq 0$)
	\item If H buys Stripped instead of Full, IC is violated $\Rightarrow$ menu is suboptimal
\end{enumerate}
