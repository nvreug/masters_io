\begin{frame}{Recap: The demographic model}
	\begin{wideitemize}
		\item Last time: the demographic interaction model
		\begin{align*}
			u_{ijt} = \underbrace{x_{jt} \beta + \alpha p_{jt} + \xi_{jt}}_{\delta_{jt}} + \underbrace{\sum_{k,\ell} \pi_{k\ell} D_{i\ell} x_{jt,k}}_{\mu_{ijt}} + \varepsilon_{ijt}
		\end{align*}
		\item $\delta_{jt}$: mean utility (common to all consumers)
		\item $\mu_{ijt}$: individual deviation based on demographics $D_i$
		\item \textbf{Key insight:} Demographics $\Rightarrow$ heterogeneous preferences $\Rightarrow$ realistic substitution
		\item \textbf{Today:} Estimate this model using pyblp
	\end{wideitemize}
\end{frame}

	\end{wideitemize}

\begin{frame}{pyblp workflow}

\begin{frame}{Model 1: Basic logit specification}
	\begin{wideitemize}
		\item \textbf{Utility:}
		\begin{align*}
			u_{ijt} = \beta_1 \cdot \text{hpwt}_{jt} + \beta_2 \cdot \text{air}_{jt} + \beta_3 \cdot \text{mpd}_{jt} + \beta_4 \cdot \text{space}_{jt} + \alpha \cdot p_{jt} + \xi_{jt} + \varepsilon_{ijt}
		\end{align*}
		\item \textbf{Parameters to estimate:}
		\begin{wideitemize}
			\item $\beta_1, \beta_2, \beta_3, \beta_4$: tastes for characteristics
			\item $\alpha$: price sensitivity (expect $\alpha < 0$)
		\end{wideitemize}
		\item \textbf{Note:} All consumers have the same preferences (no heterogeneity)
	\end{wideitemize}
\end{frame}

\begin{frame}[fragile]{Estimating Model 1}
\begin{wideitemize}
	\item Define the formulation and solve
\end{wideitemize}
\vspace{5pt}
\begin{pycode}
# Define formulation: characteristics + price (no constant)
formulation = pyblp.Formulation('0 + hpwt + air + mpd + space + prices')

# Create problem (pyblp detects instruments automatically)
problem = pyblp.Problem(formulation, product_data)

# Estimate using 2SLS
results = problem.solve()
\end{pycode}
\end{frame}

\begin{frame}[fragile]{Working with results}
	\begin{wideitemize}
		\item \textbf{Print summary:} shows coefficients, std errors, diagnostics
	\end{wideitemize}
\begin{pycode}
print(results)  # displays formatted table
\end{pycode}
	\vspace{8pt}
	\begin{wideitemize}
		\item \textbf{Access specific attributes:}
	\end{wideitemize}
\begin{pycode}
results.beta          # linear coefficients (numpy array)
results.beta_se       # standard errors
results.xi            # estimated unobservables (xi_jt)
\end{pycode}
\end{frame}

\begin{frame}[fragile]{Working with results: post-estimation}
	\begin{wideitemize}
		\item \textbf{Compute elasticities:}
	\end{wideitemize}
\begin{pycode}
elasticities = results.compute_elasticities()
# Returns matrix: elasticities[j,k] = % change in q_j from 1% price change in k

own_price_elasticities = np.diag(elasticities)  # extract diagonal
\end{pycode}
	\vspace{8pt}
	\begin{wideitemize}
		\item Own-price elasticity: $\eta_{jj}$ on diagonal
		\item Cross-price elasticity: $\eta_{jk}$ off diagonal
		\item \texttt{np.diag()} extracts the diagonal of a matrix
	\end{wideitemize}
\end{frame}

\begin{frame}{Model 2: Adding demographic interactions}
	\begin{wideitemize}
		\item \textbf{Utility with income interactions:}
		\begin{align*}
			u_{ijt} = \delta_{jt} + \underbrace{(\pi_{\text{inc}} \cdot \text{income}_i) \cdot p_{jt}}_{\mu_{ijt}} + \varepsilon_{ijt}
		\end{align*}
		\item Where $\delta_{jt} = \beta_1 \cdot \text{hpwt}_{jt} + ... + \alpha_0 \cdot p_{jt} + \xi_{jt}$
		\item \textbf{New parameter:} $\pi_{\text{inc}}$ (income $\times$ price)
		\begin{wideitemize}
			\item If $\pi_{\text{inc}} > 0$: high-income less price-sensitive
		\end{wideitemize}
		\item \textbf{Linear:} $\beta_1, ..., \alpha_0$ (in $\delta_{jt}$) \quad \textbf{Nonlinear:} $\pi_{\text{inc}}$ (in $\mu_{ijt}$)
	\end{wideitemize}
\end{frame}

\begin{frame}[fragile]{Estimating Model 2}
\begin{pycode}
# Scale income to $100,000s for numerical stability
agent_data['income'] = agent_data['income'] / 100

X1 = pyblp.Formulation('0 + hpwt + air + mpd + space + prices')
X2 = pyblp.Formulation('0 + prices')  # interacts with demographics
agent_formulation = pyblp.Formulation('0 + income')

problem = pyblp.Problem(
    (X1, X2), product_data, agent_formulation, agent_data
)
results = problem.solve(sigma=0, pi=np.array([[0.5]]))
\end{pycode}
\end{frame}

\begin{frame}{Understanding the solve command}
	\begin{wideitemize}
		\item \colorbox{gray!20}{\texttt{results = problem.solve(sigma=0, pi=np.array([[0.5]]))}}
		\item \textbf{numpy} (\texttt{np}): Python library for numerical arrays
		\begin{wideitemize}
			\item \texttt{np.array([[0.5]])} creates a matrix with value 0.5
		\end{wideitemize}
		\item \textbf{sigma}: \texttt{sigma=0} means no random coefficients
		\begin{wideitemize}
			\item We use observed demographics only
		\end{wideitemize}
		\item \textbf{pi}: Demographic interaction parameters ($\pi$)
		\begin{wideitemize}
			\item Starting values for the optimizer (initial guess)
			\item pyblp finds the best values via GMM
		\end{wideitemize}
		\item \textbf{results}: Object containing all estimation output
	\end{wideitemize}
\end{frame}

\begin{frame}{Worked example: Interpret this output (solution)}
	\begin{wideitemize}
		\item \textbf{Problem:} Price coefficient is positive (+0.05)!
		\begin{wideitemize}
			\item Higher prices $\rightarrow$ higher utility. Economically nonsensical.
		\end{wideitemize}
		\item \textbf{Why might this happen?}
		\begin{wideenumerate}
			\item Weak instruments: IVs don't predict prices well
			\item Wrong signs: Maybe you forgot a negative sign somewhere
			\item Endogeneity still present: IVs are correlated with $\xi$
		\end{wideenumerate}
		\item \textbf{Fix:} Check first-stage F-stat, try different IVs, verify data
	\end{wideitemize}
\end{frame}

\begin{frame}{Common pyblp errors and how to fix them}
	\begin{wideitemize}
		\item \textbf{``Singular matrix'' error:}
		\begin{wideitemize}
			\vspace{3pt}
			\item Likely: collinear variables in formulation
			\item Fix: Remove redundant variables
		\end{wideitemize}
		\item \textbf{Price coefficient positive or near zero:}
		\begin{wideitemize}
			\vspace{3pt}
			\item Likely: weak instruments
			\item Fix: Check first-stage F-stat; try different IVs
		\end{wideitemize}
		\item \textbf{Huge standard errors:}
		\begin{wideitemize}
			\vspace{3pt}
			\item Likely: weak identification or too few observations
			\item Fix: Aggregate to fewer markets; add more IVs
		\end{wideitemize}
	\end{wideitemize}
\end{frame}

\begin{frame}{How to check your results make sense}
	\begin{wideenumerate}
		\item \textbf{Price coefficient negative?} (Required!)
		\item \textbf{Elasticities reasonable?}
		\begin{wideitemize}
			\vspace{3pt}
			\item Own-price: typically $-2$ to $-10$ for durables
			\item Cross-price: positive but small
		\end{wideitemize}
		\item \textbf{Markups reasonable?}
		\begin{wideitemize}
			\vspace{3pt}
			\item Cars: typically 15-30\%
			\item Cereal: typically 30-50\%
		\end{wideitemize}
		\item \textbf{First-stage F-stat $> 10$?} (IV relevance)
		\item \textbf{Coefficients stable across specifications?}
	\end{wideenumerate}
\end{frame}
