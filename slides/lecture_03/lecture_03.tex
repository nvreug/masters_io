\begin{frame}{Recap: The IIA problem}
	\begin{wideitemize}
		\item Last time: basic logit gives elasticities
		\begin{align*}
			\eta_{jj} = \alpha p_{jt} (1 - s_{jt}), \qquad \eta_{jk} = -\alpha p_{kt} s_{kt}
		\end{align*}
		\item \textbf{Problem:} Cross-price elasticity $\eta_{jk}$ doesn't depend on $j$!
		\item All products have the \textit{same} cross-elasticity with product $k$
		\item Example: BMW raises price $\Rightarrow$ same fraction go to Mercedes as to Honda Civic
		\item This is unrealistic: BMW and Mercedes are closer substitutes
		\item \textbf{Today:} Fix this with demographic interactions
	\end{wideitemize}
\end{frame}

\begin{frame}{The demographic interaction model}
	\begin{wideitemize}
		\item We estimate a version of \textbf{Berry, Levinsohn \& Pakes (1995)} --- ``BLP''
		\item Allow preferences to vary across consumers:
		\begin{align*}
			u_{ijt} = x_{jt} \beta_i + \alpha_i p_{jt} + \xi_{jt} + \varepsilon_{ijt}
		\end{align*}
		\item $\beta_i$, $\alpha_i$: consumer $i$'s taste for characteristics and price
		\item \textbf{Key question:} What drives heterogeneity in $\beta_i$ and $\alpha_i$?
		\item \textbf{Our approach:} Observed demographics
		\begin{align*}
			\beta_{ik} = \beta_k + \sum_{\ell=1}^L \gamma_{k\ell} D_{i\ell}, \qquad \alpha_i = \alpha + \sum_{\ell=1}^L \gamma_{\alpha\ell} D_{i\ell}
		\end{align*}
		\item $D_{i\ell}$: demographic $\ell$ for consumer $i$ (income, age, family size, etc.)
	\end{wideitemize}
\end{frame}

\begin{frame}{Rewriting in terms of mean utility}
	\begin{wideitemize}
		\item Substituting the expressions for $\beta_{ik}$ and $\alpha_i$:
		\begin{align*}
			u_{ijt} = \underbrace{x_{jt} \beta + \alpha p_{jt} + \xi_{jt}}_{\delta_{jt} \text{ (mean utility)}} + \underbrace{\sum_{k,\ell} \gamma_{k\ell} D_{i\ell} x_{jt,k}}_{\mu_{ijt}} + \varepsilon_{ijt}
		\end{align*}
		\item \textbf{Mean utility} $\delta_{jt}$: part common to all consumers
		\begin{wideitemize}
			\item Contains ``\textbf{linear parameters}'' $(\beta, \alpha)$ --- enter linearly in $\delta$
		\end{wideitemize}
		\item \textbf{Individual deviation} $\mu_{ijt}$: how consumer $i$ differs from average
		\begin{wideitemize}
			\item Contains ``\textbf{nonlinear parameters}'' $(\gamma_{k\ell})$ --- enter nonlinearly via shares
		\end{wideitemize}
		\item Note: $\alpha < 0$ (higher price $\rightarrow$ lower utility)
	\end{wideitemize}
\end{frame}

\begin{frame}{From utility to demand}
	\begin{wideitemize}
		\item \textbf{Individual choice probability} (with T1EV errors):
		\begin{align*}
			s_{ijt} = \frac{\exp(\delta_{jt} + \mu_{ijt})}{1 + \sum_{k=1}^J \exp(\delta_{kt} + \mu_{ikt})}
		\end{align*}
		\item \textbf{Market share} (aggregate over consumers):
		\begin{align*}
			s_{jt} = \frac{1}{I} \sum_{i=1}^{I} s_{ijt}
		\end{align*}
		\item This is \textbf{demand}: the fraction of consumers who buy product $j$
		\item Key: shares depend on mean utilities $\delta$ and demographics $D_i$
	\end{wideitemize}
\end{frame}

\begin{frame}{Examples of demographic interactions}
	\begin{wideitemize}
		\item Recall: $\mu_{ijt} = \sum_{k,\ell} \gamma_{k\ell} D_{i\ell} x_{jt,k}$ captures preference heterogeneity
		\item Effective price coefficient for consumer $i$: $\alpha_i = \alpha + \gamma_{\text{inc} \times p} \cdot \text{income}_i$
		\item \textbf{Question:} High-income consumers are typically \textit{less} price-sensitive.
		\begin{wideitemize}
			\vspace{3pt}
			%\item We need $\alpha_i$ to be less negative for high-income consumers
			\item Given $\alpha < 0$, what sign should $\gamma_{\text{inc} \times p}$ have?
		\end{wideitemize}
	\end{wideitemize}

\end{frame}

\begin{frame}{Where do demographics come from?}
	\begin{wideitemize}
		\item Two scenarios:
		\begin{wideenumerate}
			\vspace{5pt}
			\item \textbf{Individual-level data:} Observe $D_i$ for each consumer
			\begin{wideitemize}
				\item Survey data, loyalty card data
			\end{wideitemize}
			\item \textbf{Market-level data:} Know distribution of $D_i$ in each market
			\begin{wideitemize}
				\item Census data, Current Population Survey
				\item This is more common in practice
			\end{wideitemize}
		\end{wideenumerate}
		\item pyblp handles both cases
	\end{wideitemize}
\end{frame}

\begin{frame}{How do we estimate this model?}
	\begin{wideitemize}
		\item Caveat: this is one algorithm, many variations exist
		\item \textbf{Two-step procedure:}
		\begin{wideenumerate}
			\vspace{5pt}
			\item \textbf{Estimate $\gamma$ and $\delta_{jt}$:}
			\begin{wideitemize}
				\item Maximum likelihood with $\delta_{jt}$ as product fixed effects
				\item Jointly estimates demographic coefficients $\gamma$
			\end{wideitemize}
			\item \textbf{Recover $\beta$ and $\alpha$:}
			\begin{wideitemize}
				\item Run 2SLS on $\delta_{jt} = x_{jt}\beta + \alpha p_{jt} + \xi_{jt}$
				\item Need instruments because price is endogenous
			\end{wideitemize}
		\end{wideenumerate}
		\item \textbf{Key insight:} First step gives us $\delta_{jt}$, then step 2 is linear regression with endogeneity (which we know how to handle!)
	\end{wideitemize}
\end{frame}

\begin{frame}{Elasticities with demographic interactions}
	\begin{wideitemize}
		\item \textbf{Elasticities with heterogeneous consumers:}
		\[  \eta_{jkt} =  \left\{
		\begin{array}{ll}
		\frac{p_{jt}}{s_{jt}} \sum_i \alpha_i s_{ijt} (1-s_{ijt}) & \text{if } j=k \\[8pt]
		-\frac{p_{kt}}{s_{jt}} \sum_i \alpha_i s_{ijt} s_{ikt} & \text{otherwise} \\
		\end{array}
		\right. \]
		\item $s_{ijt}$: probability that consumer $i$ purchases product $j$
		\item \textbf{Key insight:} Cross-elasticity now depends on $\sum_i s_{ijt} s_{ikt}$
		\begin{wideitemize}
			\item Products that attract \textit{similar consumers} have higher cross-elasticity
			\item BMW and Mercedes attract similar (high-income) consumers $\Rightarrow$ closer substitutes
		\end{wideitemize}
		\item Correlation in $\mu_{ijt}$ and $\mu_{ikt}$ drives realistic substitution patterns
	\end{wideitemize}
\end{frame}

\begin{frame}{Limitations: Demographics vs Mixed Logit}
	\begin{wideitemize}
		\item \textbf{Demographics model} (what we use):
		\begin{wideitemize}
			\vspace{5pt}
			\item Heterogeneity only from observed characteristics
			\item IIA still holds within demographic groups
			\item Simple and tractable
		\end{wideitemize}
		\item \textbf{Mixed logit / random coefficients} (advanced):
		\begin{wideitemize}
			\vspace{5pt}
			\item Heterogeneity from unobserved factors too
			\item Fully relaxes IIA
			\item Computationally intensive (requires simulation)
		\end{wideitemize}
		\item For this course: demographics are sufficient
		\item Know that mixed logit exists for research
	\end{wideitemize}
\end{frame}

		\begin{wideitemize}

			\item Standard errors

\begin{frame}{pyblp workflow}
	\begin{wideenumerate}
		\item \textbf{Set up data:} products, markets, shares, characteristics

		\item \textbf{Solve:} estimate the model
		\item \textbf{Extract results:} coefficients, standard errors, elasticities
	\end{wideenumerate}

\begin{frame}{Model 1: Basic logit specification}
	\begin{wideitemize}
		\item \textbf{Utility:}
		\begin{align*}
			u_{ijt} = \beta_1 \cdot \text{hpwt}_{jt} + \beta_2 \cdot \text{air}_{jt} + \beta_3 \cdot \text{mpd}_{jt} + \beta_4 \cdot \text{space}_{jt} + \alpha \cdot p_{jt} + \xi_{jt} + \varepsilon_{ijt}
		\end{align*}
		\item \textbf{Parameters to estimate:}
		\begin{wideitemize}
			\item $\beta_1, \beta_2, \beta_3, \beta_4$: tastes for characteristics
			\item $\alpha$: price sensitivity (expect $\alpha < 0$)
		\end{wideitemize}
		\item \textbf{Note:} All consumers have the same preferences (no heterogeneity)
	\end{wideitemize}
\end{frame}

\begin{wideitemize}
	\item Define the formulation and solve
\end{wideitemize}

\begin{pycode}
# Define formulation: characteristics + price (no constant)
formulation = pyblp.Formulation('0 + hpwt + air + mpd + space + prices')

# Create problem (pyblp detects instruments automatically)
problem = pyblp.Problem(formulation, product_data)

# Estimate using 2SLS
results = problem.solve()
\end{pycode}
\end{frame}

\begin{frame}{Model 2: Adding demographic interactions}
	\begin{wideitemize}
		\item \textbf{Utility with income interactions:}
		\begin{align*}
			u_{ijt} = \delta_{jt} + \underbrace{(\pi_{\text{inc}} \cdot \text{income}_i) \cdot p_{jt}}_{\mu_{ijt}} + \varepsilon_{ijt}
		\end{align*}
		\item Where $\delta_{jt} = \beta_1 \cdot \text{hpwt}_{jt} + ... + \alpha_0 \cdot p_{jt} + \xi_{jt}$
		\item \textbf{New parameter:} $\pi_{\text{inc}}$ (income $\times$ price)
		\begin{wideitemize}
			\item If $\pi_{\text{inc}} > 0$: high-income less price-sensitive
		\end{wideitemize}
		\item \textbf{Linear:} $\beta_1, ..., \alpha_0$ (in $\delta_{jt}$) \quad \textbf{Nonlinear:} $\pi_{\text{inc}}$ (in $\mu_{ijt}$)
	\end{wideitemize}
\end{frame}

\begin{frame}[fragile]{Estimating Model 2}
\begin{pycode}
# Scale income to $100,000s for numerical stability
agent_data['income'] = agent_data['income'] / 100

X1 = pyblp.Formulation('0 + hpwt + air + mpd + space + prices')
X2 = pyblp.Formulation('0 + prices')  # interacts with demographics
agent_formulation = pyblp.Formulation('0 + income')

problem = pyblp.Problem(
    (X1, X2), product_data, agent_formulation, agent_data
)
initial_pi = np.array([[0.5]])  # starting value for pi
results = problem.solve(sigma=0, pi=initial_pi)
\end{pycode}
\end{frame}

		\item Effective price coef: $\alpha_0 + \pi_{\text{inc}} \cdot \text{income}_i$
		\item Income = \$50k: $-0.728 + 0.014 \times 0.5 = -0.721$
		\item Income = \$200k: $-0.728 + 0.014 \times 2.0 = -0.700$ (less price-sensitive)
	\end{wideitemize}
\end{frame}

\begin{frame}{Worked example: Interpret this output (solution)}
	\begin{wideitemize}
		\item \textbf{Problem:} Price coefficient is positive (+0.05)!
		\begin{wideitemize}
			\item Higher prices $\rightarrow$ higher utility. Economically nonsensical.
		\end{wideitemize}
		\item \textbf{Why might this happen?}
		\begin{wideenumerate}
			\item Weak instruments: IVs don't predict prices well
			\item Wrong signs: Maybe you forgot a negative sign somewhere
			\item Endogeneity still present: IVs are correlated with $\xi$
		\end{wideenumerate}
		\item \textbf{Fix:} Check first-stage F-stat, try different IVs, verify data
	\end{wideitemize}
\end{frame}

\begin{frame}{Common pyblp errors and how to fix them}
	\begin{wideitemize}
		\item \textbf{``Singular matrix'' error:}
		\begin{wideitemize}
			\vspace{3pt}
			\item Likely: collinear variables in formulation
			\item Fix: Remove redundant variables
		\end{wideitemize}
		\item \textbf{Price coefficient positive or near zero:}
		\begin{wideitemize}
			\vspace{3pt}
			\item Likely: weak instruments
			\item Fix: Check first-stage F-stat; try different IVs
		\end{wideitemize}
		\item \textbf{Huge standard errors:}
		\begin{wideitemize}
			\vspace{3pt}
			\item Likely: weak identification or too few observations
			\item Fix: Aggregate to fewer markets; add more IVs
		\end{wideitemize}
	\end{wideitemize}
\end{frame}

\begin{frame}{Key Points}
	\vspace{11pt}
	\begin{wideenumerate}
		\item \textbf{Elasticities}: Own $= \alpha p_{jt}(1-s_{jt})$; Cross $= -\alpha p_{kt} s_{kt}$ \quad ($\alpha < 0$)
		\item \textbf{IIA problem}: Cross-elasticities don't depend on product similarity
		\item \textbf{Demographic interaction model}: $u_{ijt} = \delta_{jt} + \sum_{k,\ell} \gamma_{k\ell} D_{i\ell} x_{jt,k} + \varepsilon_{ijt}$
		\item Demographics allow \textbf{heterogeneous preferences} and help with IIA
		\item \textbf{pyblp workflow}: Data $\rightarrow$ Formulation $\rightarrow$ Problem $\rightarrow$ Solve $\rightarrow$ Interpret
		\item \textbf{Key outputs}: Coefficients (check signs), standard errors, elasticities, markups
		\item Price coefficient should be \textbf{negative}; OLS biases it toward zero
		\item This is the foundation for HW1
	\end{wideenumerate}
\end{frame}
